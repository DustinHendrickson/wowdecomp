file(GLOB_RECURSE PRIVATE_SOURCES CONFIGURE_DEPENDS "*.cpp")
wowdecomp_platform_sources(PRIVATE_SOURCES)

if (NOT WIN32)
    # Direct3D is Windows only. The OpenGL device builds on macOS and the Switch with its platform code in
    # CGxDeviceOpenGL/Mac (and Switch).
    list(FILTER PRIVATE_SOURCES EXCLUDE REGEX "/CGxDevice(D3d|D3d9Ex)/")
endif ()

add_library(Gx STATIC
    ${PRIVATE_SOURCES}
)

target_include_directories(Gx
    PUBLIC
        ${CMAKE_SOURCE_DIR}
)

if (WOWDECOMP_PLATFORM_SWITCH)
    # Mesa on the Switch exports no desktop GL symbols: every GL 1.x function in its <GL/gl.h> gets a trampoline that
    # jumps through a table filled from eglGetProcAddress once the context exists (SwitchGlLoadEntryPoints), so the
    # device code calls glBegin and friends as on the other platforms.
    file(STRINGS ${NX_PORTLIBS}/include/GL/gl.h GL_PROTOTYPES REGEX "GLAPIENTRY gl[A-Za-z0-9_]+ *\\(")
    set(GL_NAMES "")
    foreach (line ${GL_PROTOTYPES})
        string(REGEX MATCH "GLAPIENTRY (gl[A-Za-z0-9_]+)" match "${line}")
        list(APPEND GL_NAMES ${CMAKE_MATCH_1})
    endforeach ()
    list(REMOVE_DUPLICATES GL_NAMES)
    list(LENGTH GL_NAMES GL_COUNT)

    set(GL_TABLE "")
    set(GL_ENTRIES "")
    set(index 0)
    foreach (name ${GL_NAMES})
        string(APPEND GL_TABLE "    \"${name}\",\n")
        string(APPEND GL_ENTRIES "    \"GLENTRY ${name}, ${index}\\n\"\n")
        math(EXPR index "${index} + 1")
    endforeach ()

    set(GL_ENTRY_POINTS ${CMAKE_CURRENT_BINARY_DIR}/GlSwitchEntryPoints.cpp)
    file(WRITE ${GL_ENTRY_POINTS}.in "// Generated by Engine/Source/Gx/CMakeLists.txt from Mesa's <GL/gl.h>: ${GL_COUNT} entry points.

#include \"Engine/Source/Gx/CGxDeviceOpenGL/Switch/GlSwitch.h\"

extern \"C\" __attribute__((visibility(\"hidden\"))) void* g_switchGlEntryPoints[${GL_COUNT}];
void* g_switchGlEntryPoints[${GL_COUNT}];

static const char* const s_names[${GL_COUNT}] = {
${GL_TABLE}};

void SwitchGlLoadEntryPoints() {
    for (int i = 0; i < ${GL_COUNT}; i++) {
        g_switchGlEntryPoints[i] = SwitchGlGetProcAddress(s_names[i]);
    }
}

// Each entry point loads its slot and branches there, leaving the arguments in their registers.
__asm__(
    \".text\\n\"
    \".macro GLENTRY name, index\\n\"
    \".global \\\\name\\n\"
    \".type \\\\name, %function\\n\"
    \".p2align 2\\n\"
    \"\\\\name:\\n\"
    \"adrp x16, g_switchGlEntryPoints\\n\"
    \"add x16, x16, :lo12:g_switchGlEntryPoints\\n\"
    \"ldr x16, [x16, #(\\\\index * 8)]\\n\"
    \"br x16\\n\"
    \".size \\\\name, . - \\\\name\\n\"
    \".endm\\n\"
${GL_ENTRIES});
")
    # Only touch the source when the list changes, so the device isn't rebuilt on every configure.
    configure_file(${GL_ENTRY_POINTS}.in ${GL_ENTRY_POINTS} COPYONLY)
    target_sources(Gx PRIVATE ${GL_ENTRY_POINTS})

    # Mesa's EGL with its nouveau driver built in.
    target_link_libraries(Gx PUBLIC EGL glapi drm_nouveau)
endif ()

if (WIN32)
    target_link_libraries(Gx
        PRIVATE
            d3d9.lib
            nvapi
            opengl32.lib
    )
endif ()
